```jolt title="transformer.json"
{
    "@items": "#valueOf($.order.items)",
    "customer": "#valueOf($.order.customer.name)->#toUpperCase()",
    "total": "@items->#sum(@x: @x.price * @x.quantity)",
    "#foreach(@x in @items) into 'lines'": [
        { "sku": "@x.sku", "qty": "@x.quantity" }
    ]
}
```
